Одно правило удара на сервере — план

Для агента: выполнять по шагам (- [ ]) через executing-plans. Спека: docs/superpowers/specs/2026-11-10-strike-rules-design.md. Карточка 8925d1.

Цель: сервер отказывает в ударе раньше срока инструмента, улучшения или оружия; клиент бьёт по тому же сроку.

Архитектура: новый модуль Shared/StrikeRules (срок удара, ритм рук, счётчик Gate). Сервисы удара спрашивают один ctx.strikeGate вместо таблицы ctx.toolHitReady. Balance.hitCooldown и Balance.HandsCooldown удаляются.

Стек: Luau (--!strict для нового модуля), Rojo, тесты tests/ServerStorage/UnitTest/Cases, npm run check, npm run test:game.

Правила студии: тест пишется первым, но в начале не запускается; один прогон — когда всё готово (хук TEST-ROUND). Коммит кода — только по слову владельца.

--!strict
-- How often a player may strike. The server refuses a strike that comes too early and the
-- client swings by the same rule, so an honest player never meets a refused strike.
local Shared = game:GetService("ReplicatedStorage").CleanSnow.Shared
local StrikeRules: any = require(Shared.StrikeRules)
local Catalog: any = require(Shared.Catalog)
local WeaponDefs: any = require(Shared.WeaponDefs)
local Harness = require(game:GetService("ServerStorage").UnitTest.Harness)

local PLAYER = 1

local function shovel(level: number?): any
	return { toolId = "StarterShovel", upgrades = { ShovelSpeed = level or 0 } }
end

return function(t: Harness.Context)
	t.test("an honest shovel rhythm passes, even when two packets arrive together after a lag", function()
		local gate = StrikeRules.newGate()
		local interval = StrikeRules.interval(shovel())
		-- Strikes at 0, I and 2I; the next two are late and arrive together at 4I.
		for _, at in { 0, interval, 2 * interval, 4 * interval, 4 * interval } do
			t.expect.truthy(gate:try(PLAYER, interval, 100 + at))
		end
	end)

	t.test("an honest player punching snow nonstop is never refused", function()
		local gate = StrikeRules.newGate()
		local hands = { toolId = "Hands", upgrades = {} }
		local swingAt, streak = 0, 0
		for _ = 1, 30 do
			local cooldown = StrikeRules.handsCooldown(streak)
			-- The client sends the strike at the contact moment, 0.4 of the way into the swing.
			t.expect.truthy(gate:try(PLAYER, StrikeRules.interval(hands), swingAt + cooldown * 0.4))
			swingAt += cooldown
			streak += 1
		end
	end)

	t.test("an auto-clicker gets no more strikes than the shovel allows", function()
		local gate = StrikeRules.newGate()
		local interval = StrikeRules.interval(shovel())
		local accepted = 0
		for click = 0, 29 do -- six clicks a second for five seconds
			if gate:try(PLAYER, interval, click / 6) then
				accepted += 1
			end
		end
		t.expect.truthy(accepted <= 5 / interval + 2)
		t.expect.truthy(accepted >= 5 / interval - 1)
	end)

	t.test("each tool and weapon strikes at its own pace from the catalog", function()
		local upgrade = Catalog.Upgrades.ShovelSpeed
		t.expect.near(StrikeRules.interval(shovel(0)), upgrade.effects[1])
		t.expect.near(StrikeRules.interval(shovel(upgrade.maxLevel)), upgrade.effects[upgrade.maxLevel + 1])
		t.expect.near(StrikeRules.interval(shovel(), WeaponDefs.IceAxe), WeaponDefs.IceAxe.cooldown)
		t.expect.near(StrikeRules.interval(shovel(), WeaponDefs.SnowVacuum), WeaponDefs.SnowVacuum.tick)
	end)
end

Задача 2: модуль StrikeRules

Файлы: создать src/ReplicatedStorage/CleanSnow/Shared/StrikeRules.luau; в src/ReplicatedStorage/SnowCleaningDemo/SnowBlockWorld.luau удалить Blocks.StrikeCooldown (строки 292–294), числа ритма остаются.

  • Записать модуль:
--!strict
-- How often a player may strike: one rule for the server check and the client swing.
local Shared = script.Parent
local Catalog: any = require(Shared.Catalog)
local Balance: any = require(Shared.Balance)
local Blocks: any = require(Shared.Parent.Parent:WaitForChild("SnowCleaningDemo"):WaitForChild("SnowBlockWorld"))

local StrikeRules = {}

-- Float sums of intervals drift by a hair; a strike exactly on time must still pass.
local EPSILON = 1e-6

export type Gate = {
	try: (self: Gate, userId: number, interval: number, now: number) -> boolean,
	forget: (self: Gate, userId: number) -> (),
}

local function isShovel(profile: any): boolean
	local item = Catalog.Items[profile.toolId]
	return item ~= nil and item.kind == "Tool" and profile.toolId ~= "Hands" and profile.toolId ~= "SnowVacuum"
end

-- Shortest time between two strikes the server accepts.
function StrikeRules.interval(profile: any, weapon: any?): number
	if weapon then
		return weapon.tick or weapon.cooldown
	end
	if profile.toolId == "SnowVacuum" then
		return Balance.VacuumCooldown
	end
	if isShovel(profile) then
		return Balance.upgradeValue(profile, "ShovelSpeed")
	end
	return Blocks.MinCooldown
end

-- Hands swing faster within a streak, down to MinCooldown.
function StrikeRules.handsCooldown(streak: number?): number
	local steps = math.min(streak or 0, Blocks.StreakMax)
	return Blocks.Cooldown - (Blocks.Cooldown - Blocks.MinCooldown) * steps / Blocks.StreakMax
end

-- GCRA: each strike moves the player's next due moment by one interval. One spare strike lets
-- two packets bunched by the network through, while the average rate stays at 1/interval.
function StrikeRules.newGate(): Gate
	local due: { [number]: number } = {}
	local gate = {}
	function gate.try(_self: Gate, userId: number, interval: number, now: number): boolean
		local at = math.max(due[userId] or now, now)
		if at - now > interval + EPSILON then
			return false
		end
		due[userId] = at + interval
		return true
	end
	function gate.forget(_self: Gate, userId: number)
		due[userId] = nil
	end
	return (gate :: any) :: Gate
end

return StrikeRules
