Снег удаляется красиво: «снежки в ведро» — план

For agentic workers: REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (- [ ]) syntax for tracking.

Goal: Снятый снег распадается на круглые снежки, 1–3 из них летят в ведро HUD, пылесос втягивает снежки, огонь растапливает их в пар, золотой заливки экрана на сломе нет, Low легче High.

Architecture: Сервер добавляет косметическое effect.Source в пакет удара. Рецепт SnowBreakRecipe по качеству, «меньше движения» и способу решает числа и стиль (Hop/Suck/Melt). Пулы SnowBreakBurst (снежки из нового меша Ball) и новый пул SnowCollectFlight (точки на экране) играют рецепт. MapSnowFeedback связывает удар, рецепт, сопло пылесоса и ведро.

Tech Stack: Luau --!strict, Rojo, тесты tests/ServerStorage/UnitTest/Cases/*_Test.luau через npm run test:game -- <фильтр>, проверка npm run check, живые кадры npm run play -- <сценарий>.

Спека: docs/superpowers/specs/2026-11-15-snow-removal-vfx-design.md. Карточка d53b16. План проверен feature-plan-critic (круг 1: fail, 3 высоких и 6 средних замечаний — учтены в этой версии).

  • Studio: окна не открывать. Окно владельца (NaLovi.rbxl) и чужие окна не трогать.
  • Git не трогать: ни add, ни commit, ни stash. Коммитит оркестратор.
  • Править только свои файлы из раздела «Файлы» своей задачи. Нужен чужой файл — остановиться и написать оркестратору.

-- snowballs of an own strike fly in an arc from the struck snow to the bucket on the HUD and the bucket bumps when

	step: (now: number) -> (),
	active: () -> number,
	clear: () -> (),

function M.new(owner: Owner): Api

	screen.Name = "SnowCollectFlight"
	screen.IgnoreGuiInset = true
	screen.ResetOnSpawn = false

		local corner = Instance.new("UICorner"); corner.CornerRadius = UDim.new(0.5, 0); corner.Parent = frame
		local stroke = Instance.new("UIStroke"); stroke.Color = OUTLINE; stroke.Thickness = 2; stroke.Parent = frame
		frame.Parent = screen
		dots[index] = { Frame = frame, Start = nil, From = Vector2.zero, Lead = false }
	end
	screen.Parent = owner.Gui
	local cursor = 0
	local bumpAt: number? = nil
	local connection: RBXScriptConnection? = nil
	local destroyed = false

	local function project(point: Vector3): Vector2?
		local mapper = owner.Project
		if mapper then return mapper(point) end
		local camera = workspace.CurrentCamera
		if not camera then return nil end
		local at, visible = camera:WorldToViewportPoint(point)
		return if visible then Vector2.new(at.X, at.Y) else nil
	end
	local function aim(): Vector2?
		local bucket = owner.Bucket()
		if not bucket or not bucket.Visible then return nil end
		return bucket.AbsolutePosition + bucket.AbsoluteSize * 0.5
	end
	-- The icon and its dark outline copy beside it bump together; an icon with its own UIScale is left alone.
	local function scales(): { UIScale }
		local bucket = owner.Bucket()
		if not bucket then return {} end
		local out = {}
		local ink = bucket.Parent and bucket.Parent:FindFirstChild("ReferenceInk_" .. bucket.Name)
		for _, item in { bucket, ink } do
			if not item then continue end
			local scale = item:FindFirstChild("CollectBump")
			if not scale then
				if item:FindFirstChildOfClass("UIScale") then continue end
				local made = Instance.new("UIScale")
				made.Name = "CollectBump"
				made.Parent = item
				scale = made
			end
			if scale and scale:IsA("UIScale") then table.insert(out, scale) end
		end
		return out
	end

	local api = {} :: Api
	local function active(): number
		local n = if bumpAt then 1 else 0
		for _, dot in dots do if dot.Start then n += 1 end end
		return n
	end
	function api.step(now: number)
		local goal = aim()
		for _, dot in dots do
			local start = dot.Start
			if not start then continue end
			local t = (now - start) / FLY
			if t < 0 then dot.Frame.Visible = false; continue end
			if t >= 1 or not goal then
				dot.Start = nil
				dot.Frame.Visible = false
				if goal and dot.Lead and not bumpAt then bumpAt = now end
				continue
			end

			local at = dot.From:Lerp(goal, e) - Vector2.new(0, math.sin(t * math.pi) * ARC)
			local size = SIZE * (1 - 0.4 * t)
			dot.Frame.Position = UDim2.fromOffset(at.X, at.Y)
			dot.Frame.Size = UDim2.fromOffset(size, size)
			dot.Frame.Visible = true
		end
		local started = bumpAt
		if started then
			local t = (now - started) / BUMP_TIME
			local value = if t >= 1 or t < 0 then 1 else 1 + (BUMP_SCALE - 1) * math.sin(t * math.pi)
			for _, scale in scales() do scale.Scale = value end
			if t >= 1 then bumpAt = nil end
		end
	end
	local function run()
		if connection or destroyed then return end
		connection = RunService.Heartbeat:Connect(function()
			api.step(os.clock())
			if active() == 0 and connection then
				connection:Disconnect()
				connection = nil
			end
		end)
	end
	function api.fly(from: Vector3, count: number)
		if destroyed or owner.Reduced() or count <= 0 then return end
		local origin = project(from)
		if not origin or not aim() then return end
		local now = os.clock()
		for index = 1, math.min(count, POOL) do
			cursor = cursor % POOL + 1
			local dot = dots[cursor]
			dot.Start = now + START + (index - 1) * STAGGER
			dot.From = origin + Vector2.new((index - 2) * 14, 0)
			dot.Lead = index == 1
		end
		run()
	end
	function api.bump(delay: number)
		if destroyed or owner.Reduced() or not aim() then return end
		bumpAt = os.clock() + delay
		run()
	end
	api.active = active
	function api.clear()
		for _, dot in dots do
			dot.Start = nil
			dot.Frame.Visible = false
		end
		bumpAt = nil
		for _, scale in scales() do scale.Scale = 1 end
	end
	function api.destroy()
		destroyed = true
		if connection then
			connection:Disconnect()
			connection = nil
		end
		-- Only what this module made: the HUD may be gone already, and nothing new is made on the way out.
		local bucket = owner.Bucket()
		for _, item in { bucket, bucket and bucket.Parent and bucket.Parent:FindFirstChild("ReferenceInk_" .. bucket.Name) } do
			local scale = item and item:FindFirstChild("CollectBump")
			if scale and scale:IsA("UIScale") then scale.Scale = 1 end
		end
		screen:Destroy()
	end
	return api
end
