For agentic workers: REQUIRED SUB-SKILL: Use subagent-driven-development (recommended) or executing-plans to implement this plan task-by-task. Steps use checkbox (- [ ]) syntax for tracking.

Goal: Санта, залы аркад, убежище Гринча, голем и хоккей считают свои минуты от конца туториала, а не от старта забега: Санта через 10 с, дальше каждые 5 минут, зал 2 — через 10 минут после хоккея и стоит 5 минут.

Architecture: Новый серверный модуль RunClock раз в такт смотрит на участников забега и ставит сессии activityStartedAt (на ctx.clock(), как startedAt), когда у всех стоит tutorialDone в кошельке мира; не позже WaitMax; в забеге, где туториала ни у кого не было, — startedAt + NoTutorial. Активности вместо session.startedAt берут RunClock.origin(session, fromRun); fromRun — сценарий Studio задал своё *After, тогда счёт от старта забега, как раньше (сценарии не ломаются). Dev-панель двигает оба времени.

--!strict
local RunClock: any = require(game:GetService("ServerScriptService").CleanSnow.RunClock)
local Harness = require(game:GetService("ServerStorage").UnitTest.Harness)

local RULES = { WaitMax = 600, NoTutorial = 30 }

-- players[id]: the member's wallet view (nil: not loaded); inRun[id]: still in this run.
local function world(members: { number }): any
	local session: any = { id = "ClockTest", phase = "Active", startedAt = 100, members = members }
	local state: any = { players = {}, inRun = {} }
	for _, id in members do state.inRun[id] = true end
	local clock = RunClock.new({
		wallets = { get = function(id: number): any return state.players[id] end },
		sessions = { getForUser = function(id: number): any return if state.inRun[id] then session else nil end },
	}, { rules = RULES })
	clock.onSessionStarted(session)
	return { clock = clock, session = session, state = state, tick = function(now: number) clock.tick(0.1, now) end }
end

return function(t: Harness.Context)
	local expect = t.expect

	t.test("solo: waits through the tutorial and starts when it is done (the lesson or SKIP)", function()
		local m = world({ 1 })
		m.state.players[1] = { tutorialDone = false }
		m.tick(150)
		m.tick(300)
		expect.equal(m.session.activityStartedAt, nil)
		m.state.players[1] = { tutorialDone = true }
		m.tick(320)
		expect.equal(m.session.activityStartedAt, 320)
		m.tick(400)
		expect.equal(m.session.activityStartedAt, 320)
	end)

	t.test("a world continued after the tutorial starts NoTutorial seconds into the run", function()
		local m = world({ 1, 2 })
		m.state.players[1] = { tutorialDone = true }
		m.state.players[2] = { tutorialDone = true }
		m.tick(110)
		expect.equal(m.session.activityStartedAt, nil)
		m.tick(131)
		expect.equal(m.session.activityStartedAt, 130)
	end)

	t.test("a party waits for everyone in the tutorial, a wallet not loaded yet holds it", function()
		local m = world({ 1, 2 })
		m.state.players[1] = { tutorialDone = false }
		m.tick(150)
		m.state.players[1] = { tutorialDone = true }
		m.tick(200)
		expect.equal(m.session.activityStartedAt, nil)
		m.state.players[2] = { tutorialDone = false }
		m.tick(250)
		m.state.players[2] = { tutorialDone = true }
		m.tick(260)
		expect.equal(m.session.activityStartedAt, 260)
	end)

	t.test("no longer than WaitMax, and a member who left the run does not hold it", function()
		local m = world({ 1, 2 })
		m.state.players[1] = { tutorialDone = false }
		m.state.players[2] = { tutorialDone = false }
		m.tick(699)
		expect.equal(m.session.activityStartedAt, nil)
		m.tick(700)
		expect.equal(m.session.activityStartedAt, 700)
		local n = world({ 1, 2 })
		n.state.players[1] = { tutorialDone = false }
		n.state.players[2] = { tutorialDone = false }
		n.tick(150)
		n.state.players[1] = { tutorialDone = true }
		n.state.inRun[2] = nil
		n.tick(160)
		expect.equal(n.session.activityStartedAt, 160)
	end)

	t.test("a player who joins after the start does not move the clock; begin starts it only once", function()
		local m = world({ 1 })
		m.state.players[1] = { tutorialDone = false }
		m.tick(150)
		m.state.players[1] = { tutorialDone = true }
		m.tick(170)
		table.insert(m.session.members, 9)
		m.state.inRun[9] = true
		m.state.players[9] = { tutorialDone = false }
		m.tick(200)
		expect.equal(m.session.activityStartedAt, 170)
		m.clock.begin(m.session, 500)
		expect.equal(m.session.activityStartedAt, 170)
		local n = world({ 1 })
		n.clock.begin(n.session, 120)
		expect.equal(n.session.activityStartedAt, 120)
	end)

	t.test("origin: the clock's start, the run's start for a Studio scenario, nil while it waits", function()
		local session: any = { startedAt = 100 }
		expect.equal(RunClock.origin(session), nil)
		expect.equal(RunClock.origin(session, true), 100)
		session.activityStartedAt = 250
		expect.equal(RunClock.origin(session), 250)

--!strict
-- Card 435b9c (Leonid's decisions 22.11.2026, spec docs/superpowers/specs/2026-11-22-santa-after-tutorial-design.md,
-- «Часы забега»): the run's activities (Santa, the arcade halls, the Grinch's hideout, the golem, the hockey rink)
-- count their minutes from the end of the tutorial, not from the run's start. The clock starts when every member still
-- in the run has the tutorial done in the run's world (the wallet's tutorialDone: the island lesson or SKIP), but no
-- later than WaitMax seconds into the run, so a slow friend does not hold everyone. A run where nobody had the
-- tutorial (a world continued after it) starts it NoTutorial seconds into the run. It never starts again: a player who
-- joins later does not move it. The clock is the session's activityStartedAt, on ctx.clock() like startedAt.
local Balance: any = require(game:GetService("ReplicatedStorage").CleanSnow.Shared.Balance)

local RunClock = {}

export type Deps = { rules: any? }

-- The time the run's activities count from; nil while the clock waits. fromRun: from the run's start instead (a Studio
-- scenario that set the activity's own After attribute keeps its old meaning).
function RunClock.origin(session: any, fromRun: boolean?): number?
	local at = if fromRun then session.startedAt else session.activityStartedAt
	return if type(at) == "number" then at else nil
end

function RunClock.new(ctx: any, deps: Deps?): any
	local rules = (deps and deps.rules) or Balance.RunClock
	-- lesson: someone of the run was seen in the tutorial.
	local runs: { [string]: { session: any, lesson: boolean } } = {}
	local api: any = {}

	function api.onSessionStarted(session: any)
		runs[session.id] = { session = session, lesson = false }
	end

	function api.onSessionClosed(session: any)
		runs[session.id] = nil
	end

	-- Starts the clock at `now` if it has not started (the Studio Dev panel raising an event before it).
	function api.begin(session: any, now: number)
		if session.activityStartedAt == nil then session.activityStartedAt = now end
	end

	function api.tick(_dt: number, now: number)
		for _, run in runs do
			local session = run.session
			local started = session.startedAt
			if session.activityStartedAt ~= nil or session.phase ~= "Active" or type(started) ~= "number" then continue end
			local waiting = false
			for _, id in session.members do
				if ctx.sessions.getForUser(id) ~= session then continue end
				local player = ctx.wallets.get(id)
				if player == nil then
					waiting = true
				elseif player.tutorialDone ~= true then
					waiting = true
					run.lesson = true
				end
			end
			if now - started >= rules.WaitMax then
				session.activityStartedAt = now
			elseif not waiting and run.lesson then
				session.activityStartedAt = now
			elseif not waiting and now - started >= rules.NoTutorial then
				session.activityStartedAt = started + rules.NoTutorial
			end
		end
	end

	return api
end

return RunClock
